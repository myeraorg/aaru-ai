export default { async fetch(request, env) { const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" }; if (request.method === "OPTIONS") { return new Response(null, { headers: cors }); } try { let prompt = ""; let ratio = "1:1"; let quality = "standard"; /* ========================= GET REQUEST ========================= */ if (request.method === "GET") { const url = new URL(request.url); prompt = url.searchParams.get("prompt") || ""; ratio = url.searchParams.get("ratio") || "1:1"; quality = url.searchParams.get("quality") || "standard"; } /* ========================= POST REQUEST ========================= */ if (request.method === "POST") { const data = await request.json(); prompt = data.prompt || ""; ratio = data.ratio || "1:1"; quality = data.quality || "standard"; } /* ========================= VALIDATE PROMPT ========================= */ if (!prompt.trim()) { return new Response( JSON.stringify({ error: "Prompt is required" }), { status: 400, headers: { "Content-Type": "application/json", ...cors } } ); } /* ========================= MULTILINGUAL AI ========================= */ const messages = [ { role: "system", content: ` You are the multilingual prompt engine for an AI image generator. The user may write in ANY language. Understand the user's exact meaning, including: - Hindi - English - Bengali - Tamil - Telugu - Marathi - Gujarati - Urdu - Arabic - Japanese - Korean - Chinese - French - German - Spanish - Portuguese - Russian - and other languages. Your job is NOT to answer the user. Your job is to convert the user's request into ONE clear, detailed English prompt for an image-generation model. Preserve the user's requested: - objects - people - animals - colors - clothing - location - environment - actions - lighting - camera angle - style - mood - composition - quantity Do not change the user's intended subject. If the user gives a short prompt, expand it naturally into a useful image prompt without inventing important objects. Return ONLY the final English image prompt. Do NOT include explanations. Do NOT include quotation marks. Do NOT say that you translated the prompt. ` }, { role: "user", content: prompt } ]; /* ========================= CALL GLM ========================= */ const aiResponse = await env.AI.run( "@cf/zai-org/glm-4.7-flash", { messages: messages, temperature: 0.2, max_tokens: 512 } ); /* ========================= EXTRACT GLM RESPONSE ========================= */ let imagePrompt = ""; if ( aiResponse && aiResponse.choices && aiResponse.choices.length > 0 ) { const choice = aiResponse.choices[0]; if ( choice.message && typeof choice.message.content === "string" ) { imagePrompt = choice.message.content.trim(); } } /* Fallback */ if ( !imagePrompt && aiResponse && typeof aiResponse.response === "string" ) { imagePrompt = aiResponse.response.trim(); } /* ========================= GLM FAILURE ========================= */ if (!imagePrompt) { console.error( "GLM response:", JSON.stringify(aiResponse) ); return new Response( JSON.stringify({ error: "Multilingual prompt processing failed." }), { status: 500, headers: { "Content-Type": "application/json", ...cors } } ); } /* ========================= IMAGE SIZE ========================= */ let width = 1024; let height = 1024; if (ratio === "9:16") { width = 768; height = 1344; } if (ratio === "16:9") { width = 1344; height = 768; } /* ========================= QUALITY ========================= */ let steps = 8; if (quality === "hd") { steps = 20; } /* ========================= GENERATE IMAGE ========================= */ const image = await env.AI.run( "@cf/stabilityai/stable-diffusion-xl-base-1.0", { prompt: imagePrompt, width: width, height: height, num_steps: steps } ); /* ========================= RETURN IMAGE ========================= */ return new Response( image, { headers: { "Content-Type": "image/png", "Cache-Control": "no-store", ...cors } } ); } catch (error) { console.error( "Aaru AI Error:", error ); return new Response( JSON.stringify({ error: error && error.message ? error.message : "Unknown Worker error" }), { status: 500, headers: { "Content-Type": "application/json", ...cors } } ); } } };
